{# figure shortcode. Three kinds of `src` are supported: - remote URLs (http/https): used as-is, no metadata lookup. - absolute paths ("/img/..."): looked up under the project root. - colocated assets (a bare filename): resolved relative to the current page bundle, so `content//index.md` can reference `foo.png`. Local images get intrinsic width/height via get_image_metadata. #} {%- if src -%} {%- set is_remote = src is starting_with("http") -%} {%- set is_absolute = src is starting_with("/") -%} {%- set width = 0 -%} {%- set height = 0 -%} {%- set out_src = src -%} {%- if not is_remote -%} {%- if is_absolute -%} {%- set lookup_path = src | trim_start_matches(pat="/") -%} {%- else -%} {# colocated asset: content//index.md -> content// #} {%- set bundle_dir = page.relative_path | trim_end_matches(pat="index.md") -%} {%- set lookup_path = bundle_dir ~ src -%} {# emit an absolute URL so the image also resolves in feeds #} {%- set out_src = page.permalink ~ src -%} {%- endif -%} {%- set meta = get_image_metadata(path=lookup_path, allow_missing=true) -%} {%- if meta -%} {%- set width = meta.width -%} {%- set height = meta.height -%} {%- endif -%} {%- endif -%} {%- endif -%}
